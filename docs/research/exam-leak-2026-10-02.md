```bash
# 汇总（打印 MATERIAL 行数 / 命中数 / 按判据 / 按节点；有命中退出码 1）
PYTHONPATH=tools python -m kb_coverage.check_exam_leak

# 全量明细（stdout 206 行，字段与 §3 同源同序）
PYTHONPATH=tools python - <<'PY'
import csv, sys
sys.path.insert(0, "tools")
from kb_build.audit_material_examples import classify
rows = list(csv.DictReader(open("tools/kb_coverage/tables/material_judgments.csv", encoding="utf-8-sig")))
for r in rows:
    if (r.get("action") or "").strip() != "MATERIAL":
        continue
    fake = {"title": r.get("title") or "", "contentMarkdown": r.get("content") or "",
            "summaryMarkdown": r.get("summary") or "", "applicabilityMarkdown": r.get("applicability") or ""}
    hit, why = classify(fake)
    if hit:
        print(r["node_slug"], "|", why, "|", r["chunk_rel"], "|", r["chunk_id"] + (r.get("midx") or ""))
PY
```

> 生成方式：本文件由上述同一 `classify` 代码路径复算生成（subject 列由包内 slug→subject 映射补出），非手工誊抄。
