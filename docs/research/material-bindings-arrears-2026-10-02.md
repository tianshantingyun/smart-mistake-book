import csv, json
from pathlib import Path
REL = Path("core/data/src/main/resources/knowledge")
def load(p):
    with open(p, encoding="utf-8") as fh:
        return json.load(fh)
pack = load(REL/"moe-2025-four-subjects-v1.json")
idx = load(REL/"moe-2025-teaching-support-v2-index.json")
sidecars = [REL/n.rsplit("/",1)[-1] for n in idx["sidecars"]]
points = {(s["subject"], p["slug"]) for s in pack["subjects"] for t in s["topics"] for p in t.get("knowledgePoints") or []}
mats = {}
for sp in sidecars:
    for m in load(sp)["materials"]:
        mats.setdefault(m["slug"], []).append(m)
rows = list(csv.DictReader(open("tools/kb_build/tables/material_bindings.csv", encoding="utf-8-sig")))
c=a=mp=et=0
for r in rows:
    mc = mats.get(r["material_slug"])
    if not mc: continue
    if not (r["point_slug"] or "").strip():
        et+=1; continue
    if (r["subject"], r["point_slug"]) not in points:
        mp+=1; continue
    cur = (mc[0].get("bindings") or [{}])[0].get("knowledgeNodeId","").rsplit(":atomic:",1)[-1]
    if cur == r["point_slug"]: c+=1
    else: a+=1
print("consistent=%d arrear=%d missing_point=%d empty_target=%d" % (c,a,mp,et))
